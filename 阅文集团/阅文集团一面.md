16. 实现一个 useAxios，API 可以自己设计，讲一下大致的整个思路

``` tsx
import { useState, useEffect, useCallback, useRef } from 'react'
import axios from 'axios'

function useAxios<T>(options: UseAxiosOptions): UseAxiosReturn<T> {
    const [data, setData] = useState<T | null>(null)
    const [loading, setLoading] = useState(false)
    const [error, setError] = useState<Error | null>(null)

    const abortCtrlRef = useRef<AbortController>()
    const retryCountRef = useRef(0)
    const debounceTimerRef = useRef<number>()

    const fetch = useCallback(async (overrides?: Partial<UseAxiosOptions>) => {
        const opt = { ...options, ...overrides }
        abortCtrlRef.current?.abort()
        const ctrl = new AbortController()
        abortCtrlRef.current = ctrl

        setLoading(true)
        setError(null)
        try {
            const res = await axios({
                url: opt.url,
                method: opt.method,
                params: opt.params,
                data: opt.data,
                signal: ctrl.signal
            })
            setData(res.data)
            retryCountRef.current = 0
        } catch (err: any) {
            // 不是主动取消才处理错误
            if (!axios.isCancel(err)) {
                // 重试逻辑
                if (retryCountRef.current < (opt.retry ?? 0)) {
                    retryCountRef.current++
                    setTimeout(() => fetch(overrides), 600)
                    return
                }
                setError(err)
            }
        } finally {
            setLoading(false)
        }
    }, [options])

    const run = useCallback((overrides?: Partial<UseAxiosOptions>) => {
        if (options.debounce) {
            clearTimeout(debounceTimerRef.current)
            debounceTimerRef.current = window.setTimeout(() => fetch(overrides), options.debounce)
        } else {
            fetch(overrides)
        }
    }, [fetch, options.debounce])

    const cancel = useCallback(() => {
        abortCtrlRef.current?.abort()
    }, [])

    const reset = useCallback(() => {
        setData(null)
        setError(null)
        setLoading(false)
        retryCountRef.current = 0
    }, [])

    // autoRun 自动请求
    useEffect(() => {
        if (options.autoRun !== false) {
            run()
        }
        return () => {
            abortCtrlRef.current?.abort()
            clearTimeout(debounceTimerRef.current)
        }
    }, [run, options.autoRun])

    return { data, loading, error, run, cancel, reset }
}
```
