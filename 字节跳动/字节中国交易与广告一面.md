``` JavaScript
// 主包写的
class Scheduler {
    MAX = 2;
    runningCount = 0;  // 当前正在运行的任务计数
    queue = [];        // 等待队列，存放任务创建函数

    add(promiseCreator) {
        this.queue.push(promiseCreator);
        this.run();
    }

    run() {
        // 有等待任务，并且当前运行数 < 最大并发，才可以执行
        while(this.runningCount < this.MAX && this.queue.length > 0) {
        const task = this.queue.shift();
        this.runningCount++;
        task()
            .then(() => {
                this.runningCount--;  // 任务完成，计数减一，继续从队列拿任务执行
                this.run();
            })
        }
    }
}

const timeout = (time, value) => new Promise(resolve => {
    setTimeout(() => {
        console.log(value);
        resolve(value);
    }, time)
})

const scheduler = new Scheduler()
const addTask = (time, order) => {
    scheduler.add(() => timeout(time, order))
}

addTask(1000, '1')
addTask(500, '2')
addTask(300, '3')
addTask(400, '4')
// output: 2 3 1 4
```

6. 新增 add 功能，Schedule.add，它会返回一个 Promise，然后当这个添加的 Promise 被执行的时候，那个它返回的 Promise 会被调到，你自己加上另外一组对 return 的测试
``` JavaScript
// 主包写的
class Scheduler {
    MAX = 2;
    runningCount = 0;
    queue = [];

    add(promiseCreator) {
        return new Promise((resolve, reject) => {
        this.queue.push({
            promiseCreator,
            resolve,
            reject
        });
            this.run();
        });
    }

    run() {
        if (this.runningCount < this.MAX && this.queue.length > 0) {
        const { promiseCreator, resolve, reject } = this.queue.shift();
        this.runningCount++;
        promiseCreator()
            .then((res) => resolve(res))
            .catch((err) => reject(err))
            .finally(() => {
                this.runningCount--;
                this.run();
            });
        }
    }
}

const timeout = (time, value) => new Promise(resolve => {
    setTimeout(() => {
        console.log(value);
        resolve(value);
    }, time)
})

const scheduler = new Scheduler()
// ✅ 这里必须写 return！！

    return scheduler.add(() => timeout(time, order))
```
