28. 我现在想让你写一个闭包函数啊，这个函数就叫做 A。然后这个 A 每次执行的时候可以做一个计数器的的一个效果。比如说我 A 调用一次就返回一，第二次返回二，第三次返回三

``` JavaScript
function A() {
    // 私有计数变量，被闭包保护
    let count = 0;
    // 返回内层函数，形成闭包
    return function () {
        count++;
        console.log(count);
    }
}

// 获取计数器实例
const a = A();

a(); // 输出 1
a(); // 输出 2
a(); // 输出 3
```

29. 你平时在做这个前端开发的时候，会请求后端的接口对吧？但是你请求呢，就有可能成功也有可能失败.成功的时候我们就不说了，现在假如说失败的话，我们想要去做一个自动重试的这么一个操作。然后我们会自己去封装一个接口，比如封装一个函数，这个请求函数就叫 request。然后这个里面会给你几个参数，第一个就是 options，就是你去请求的时候去透传给这个请求函数。第二个参数会有一个 maxcount，这个就是说我在请求失败的时候我会去做自动重试，最多重试多少次。第三个参数会传一个叫 interval ，就是说我每隔多长时间去执行这样一个重试。然后这个 request 就是我们自己业务上封装的这么一个 request，这里边你可以去用呃你熟悉的请求方式去去做，或者是其他的都可以。你写一下这个这个函数。

``` JavaScript
async function request(options, maxCount, interval) {
    // 记录已经重试的次数
    let retryTimes = 0;

    // 内部循环执行请求
    async function execute() {
        try {
            // 发起网络请求，使用fetch
            const res = await fetch(options.url, options);
            // 这里我们把http状态码非2xx也当作失败，需要重试
            if (!res.ok) {
                throw new Error(`http error status: ${res.status}`);
            }
            const data = await res.json();
            return data;
        } catch (err) {
            console.error("请求失败", err.message);
            // 判断是否还可以重试：已重试次数 < 最大重试次数
            if (retryTimes < maxCount) {
                retryTimes++;
                console.log(`准备第${retryTimes}次重试，等待${interval}ms`);
                // 延迟interval毫秒之后再发起请求
                await sleep(interval);
                return execute();
            } else {
                // 达到最大重试次数，抛出错误，交给上层捕获
                throw err;
            }
        }
    }
  return execute();
}

// 工具函数：sleep延迟
function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}

// 测试用例
// maxCount=2：最多失败后重试2次；interval=1000ms间隔1秒重试
request({
    url: "/api/xxx",
    method: "GET"
}, 2, 1000)
.then(res => {
    console.log("成功结果", res);
})
.catch(e => {
    console.error("全部重试完毕依旧失败", e);
});
```
